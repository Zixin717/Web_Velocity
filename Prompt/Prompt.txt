    %% 1. 會員表 (Members)             ->（ 緊急程度：5 一定要在報告前生出來 ）
    Members {
        INT MemberID PK "會員編號"
        NVARCHAR Email "帳號/信箱"
        NVARCHAR Password "密碼"
        NVARCHAR Name "姓名"
        NVARCHAR Address "預設地址"
        VARCHAR Phone "電話"
        DATETIME CreatedAt "創建時間"
    }

    %% 2. 商品表 (Products)          ->（ 緊急程度：5 一定要在報告前生出來 ）
    Products {
        INT ProductID PK "商品編號"
        NVARCHAR Name "商品名稱"
        INT Price "價格"
        INT Stock "庫存"
        NVARCHAR Description "描述"
        NVARCHAR ImageURL "圖片路徑"
        NVARCHAR Flavor "口味"
    }


    %% 3. 購物車 (Cart) - 中繼站     ->（ 緊急程度：4 ）
    ShoppingCart {
        INT CartID PK "流水號"
        INT MemberID FK "會員編號"
        DATETIME UpdatedAt "更新時間"
    }

    %% 4. 購物車明細 (CartItems) - （ 緊急程度：4）
    CartItems {
        INT CartItemID PK "流水號"
        INT CartID FK "購物車"
        INT ProductID FK "商品編號"
        INT Quantity "數量"
        DATETIME AddedDate "更新時間"
    }

    %% 5. 訂單主表 (Orders) - （ 緊急程度：4 ）
    Orders {
        INT OrderID PK "訂單編號"
        INT MemberID FK "會員編號"
        DATETIME OrderDate "下單時間"
        INT TotalAmount "總金額"
        NVARCHAR Status "狀態"
        NVARCHAR ShippingAddress "寄送地址"
    }

    %% 6. 訂單明細表 (OrderDetails)                ->（ 緊急程度：3 先做完 5、4 再弄 3 ）
    OrderDetails {
        INT OrderDetailID PK "流水號"
        INT OrderID FK "訂單編號"
        INT ProductID FK "商品編號"
        INT Quantity "數量"
        INT UnitPrice "購買時單價"
    }

    %% --- 關聯線 ---
