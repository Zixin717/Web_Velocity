    %% 1. 會員表 (Members)
    Members {
        INT memberID PK "會員編號"
        NVARCHAR email "帳號/信箱"
        NVARCHAR password "密碼"
        NVARCHAR name "姓名"
        NVARCHAR address "預設地址"
        VARCHAR phone "電話"
        DATETIME createdAt "創建時間"
    }

    %% 2. 商品表 (Products)
    Products {
        INT productID PK "商品編號"
        NVARCHAR name "商品名稱"
        INT price "價格"
        INT stock "庫存"
        NVARCHAR description "描述"
        NVARCHAR imageURL "圖片路徑"
        NVARCHAR flavor "口味"
    }

    %% 3. 購物車 (Cart) - 中繼站
    Cart {
        INT cartID PK "流水號"
        INT memberID FK "會員編號"
        DATETIME updatedAt "更新時間"
    }

    %% 4. 購物車明細 (CartItems)
    CartItems {
        INT cartItemID PK "流水號"
        INT cartID FK "購物車"
        INT productID FK "商品編號"
        INT quantity "數量"
        DATETIME addedDate "更新時間"
    }

    %% 5. 訂單主表 (Orders)
    Orders {
        INT orderID PK "訂單編號"
        INT memberID FK "會員編號"
        DATETIME orderDate "下單時間"
        INT totalAmount "總金額"
        NVARCHAR status "狀態"
        NVARCHAR shippingAddress "寄送地址"
    }

    %% 6. 訂單明細表 (OrderDetails)
    OrderDetails {
        INT orderDetailID PK "流水號"
        INT orderID FK "訂單編號"
        INT productID FK "商品編號"
        INT quantity "數量"
        INT unitPrice "購買時單價"
    }

    %% --- 關聯線 ---
    
    %% 會員行為
    Members ||--o{ Orders : "下單"
    Members ||--|| Cart : "擁有 (1對1)"

    %% 購物車結構
    Cart ||--o{ CartItems : "包含 (1對多)"
    Products ||--o{ CartItems : "放入車中"

    %% 訂單結構 
    Orders ||--o{ OrderDetails : "包含 (1對多)"
    Products ||--o{ OrderDetails : "被購買"
